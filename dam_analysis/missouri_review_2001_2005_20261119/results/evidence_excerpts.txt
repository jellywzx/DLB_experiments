350:     def KGE(self, s, o):
351:         """
352:         Kling-Gupta Efficiency
353:         input:
354:             s: simulated
355:             o: observed
356:         output:
357:             kge: Kling-Gupta Efficiency
358:             cc: correlation
359:             alpha: ratio of the standard deviation
360:             beta: ratio of the mean
361:         """
362:         s, o = self._validate_inputs(s, o)
363:         cc = self.correlation(s, o)
364:         # Guard against constant observation (std=0) and zero-mean observation
365:         # (e.g. precipitation in a dry month). Without these guards alpha/beta
366:         # become inf and kge becomes -inf, silently polluting downstream output.
367:         o_std = o.std(dim="time")
368:         o_mean = o.mean(dim="time")
369:         alpha = xr.where(varies_along(o) & (o_std != 0), s.std(dim="time") / o_std, np.nan)
370:         beta = xr.where(o_mean != 0, s.mean(dim="time") / o_mean, np.nan)
371:         kge = 1 - ((cc - 1) ** 2 + (alpha - 1) ** 2 + (beta - 1) ** 2) ** 0.5
372:         return kge  # , cc, alpha, beta
